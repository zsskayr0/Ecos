// MEDIÇÃO DESCARTÁVEL (não commitar): conta chamadas de API ao montar Topbar + Feed e após um "notificar".
import { act, render } from "@testing-library/react";
import { MemoryRouter } from "react-router-dom";
import { describe, expect, it, vi } from "vitest";

const contagem: Record<string, number> = {};
const marca = (k: string) => { contagem[k] = (contagem[k] ?? 0) + 1; };
const atraso = <T,>(v: T) => new Promise<T>((r) => setTimeout(() => r(v), 20));

vi.mock("@/lib/api", async (orig) => {
  const real = await orig<typeof import("@/lib/api")>();
  return {
    ...real,
    feed: { obter: vi.fn(() => { marca("GET /feed"); return atraso({ items: [], next_cursor: null }); }) },
    notas: { ...real.notas, listar: vi.fn(() => { marca("GET /notas"); return atraso({ items: [], next_cursor: null }); }) },
    tarefas: { ...real.tarefas, listar: vi.fn(() => { marca("GET /tarefas"); return atraso({ items: [], next_cursor: null }); }) },
    equipes: { ...real.equipes, listarMinhas: vi.fn(() => { marca("GET /equipes"); return atraso([{ id: "E1", nome: "Casa", cargo: "dono", tipo: "pessoal", membros: 1 }]); }) },
    notificacoes: { ...real.notificacoes, naoLidas: vi.fn(() => { marca("GET /notificacoes/nao-lidas"); return atraso({ total: 0 }); }) },
  };
});
vi.mock("@/lib/auth-context", async (orig) => {
  const real = await orig<typeof import("@/lib/auth-context")>();
  const perfil = { id: "U1", nome_usuario: "d", nome: "D", equipes: [], avatar_atualizado_em: null };
  return { ...real, useAuth: () => ({ perfil, status: "autenticado" }) };
});

import { RefreshProvider, useRefreshBus } from "@/lib/refresh-bus";
import { AppUIProvider } from "@/lib/ui-context";
import { Topbar } from "@/components/layout/Topbar";
import { FeedScreen } from "@/screens/Feed/FeedScreen";

let notificarFora: () => void = () => {};
function Captura() { notificarFora = useRefreshBus().notificar; return null; }

const total = () => Object.values(contagem).reduce((a, b) => a + b, 0);
const quieto = () => act(async () => { await new Promise((r) => setTimeout(r, 400)); });

describe("medicao", () => {
  it("conta chamadas", async () => {
    render(
      <MemoryRouter><RefreshProvider><AppUIProvider><Captura /><Topbar /><FeedScreen /></AppUIProvider></RefreshProvider></MemoryRouter>,
    );
    await quieto();
    console.log("MONTAGEM Topbar+Feed (400 ms):", JSON.stringify(contagem), "total", total());
    for (const k of Object.keys(contagem)) contagem[k] = 0;
    await act(async () => { notificarFora(); });
    await quieto();
    console.log("APOS 1 notificar() (ex.: salvar uma nota):", JSON.stringify(contagem), "total", total());
    expect(true).toBe(true);
  });
});
